### Technical Milestone

Here are a few of the circuits...
```bash
module top_module (
    input clk,
    input reset,
    output OneHertz,
    output [2:0] c_enable
); //
    wire [3:0] ctr0, ctr1, ctr2;
    
    bcdcount counter0 (clk, reset, c_enable[0], ctr0[3:0]);
    bcdcount counter1 (clk, reset, c_enable[1], ctr1[3:0]);
    bcdcount counter2 (clk, reset, c_enable[2], ctr2[3:0]);
    
    
    assign c_enable[0] = ~reset;
    assign c_enable[1] = ctr0[3]&~ctr0[2]&~ctr0[1]&ctr0[0];
    assign c_enable[2] = ctr1[3]&~ctr1[2]&~ctr1[1]&ctr1[0] & c_enable[1];
    assign OneHertz = ctr2[3]&~ctr2[2]&~ctr2[1]&ctr2[0] & c_enable[2];
    
endmodule



```
