module top_module (
    input c,
    input d,
    output [3:0] mux_in
); 
    wire mux_in3;
    assign mux_in[0] = d ? 1'b1 : c;
    assign mux_in[1] = 1'b0;

    assign mux_in[2] = d ? 1'b0 : 1'b1;
    
    assign mux_in3 = d ? 1'b1 : 1'b0;
    assign mux_in[3] = c ? mux_in3 : 1'b0;

endmodule

module top_module (
    input clk,
    input resetn,
    input [1:0] byteena,
    input [15:0] d,
    output [15:0] q
);
    
    wire [15:0] d2;
    
    always @(posedge clk) begin
    
        q[7:0] = byteena[0] ? d[7:0] : q[7:0];
        q[15:8] = byteena[1] ? d[15:8] : q[15:8];
        
        q[15:0] = resetn ? q[15:0] : 16'b0000000000000000;
        
    end

endmodule
