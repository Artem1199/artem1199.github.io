```bash
module top_module (
    input clk,
    input reset,      // Synchronous active-high reset
    output [3:0] q);
    
    always @(posedge clk) begin
       
        q <= (reset) ? 4'b0000 : q + 1;
        
    end

endmodule

module top_module (
    input clk,
    input reset,        // Synchronous active-high reset
    output [3:0] q);
    
    always @(posedge clk) begin   
        q <= (reset | (q==9)) ? 4'b0000 : q + 1;  
    end
    
endmodule

module top_module (
    input clk,
    input slowena,
    input reset,
    output [3:0] q);
        
    always @(posedge clk) begin   
        
        if (reset) q <= 0;
        else if (slowena & q == 9) q <= 0;
        else if (slowena) q <= q+1;
        else q <= q;
    end

endmodule

```
