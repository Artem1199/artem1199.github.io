module top_module (
    input clk,
    input d, 
    input ar,   // asynchronous reset
    output q);
    
    always @(posedge clk or posedge ar) begin
       
        if (ar == 1'b1) q <= 1'b0;
        else if(clk == 1'b1) q <= d;
    end
    

endmodule

module top_module (
    input clk,
    input in, 
    output out);
    
    wire xorout;
    
    assign xorout = in ^ out;
    
    always @(posedge clk) begin
       out <= xorout; 
    end

endmodule

module top_module (
    input clk,
    input x,
    output z
); 
    wire xorout, andout, orout, q1,q2, q3, nq2, nq3;
    
    initial begin
       q1 = 1'b0;
       q2 = 1'b0;
       q3 = 1'b0;
    end
    
    assign xorout = q1 ^ x;
    assign andout = nq2 & x;
    assign orout = nq3 | x;
    assign z = ~(q1 | q2 | q3);
    
    always @(posedge clk) begin
       q1 <= xorout;
       
       q2 <= andout;
       nq2 <= ~andout;
       
       q3 <= orout;
       nq3 <= ~orout;
    end
                

endmodule

module top_module (
    input clk,
    input j,
    input k,
    output Q); 
    
    wire d;
    
    assign d = ~j&~k&Q|j&~k&~Q|j&~k&Q|j&k&~Q; 
    
    always @(posedge clk) begin
       
        Q <= d;
        
    end

endmodule
