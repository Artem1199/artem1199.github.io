```bash
module top_module (
    input clk,
    input reset,
    input enable,
    output [3:0] Q,
    output c_enable,
    output c_load,
    output [3:0] c_d
); //
   
    
    count4 the_counter(.clk(clk), .enable(enable), .load(c_load), .d(c_d), .Q(Q));

    
    always @(*) begin
        if (enable & ~reset) c_enable = 1;
        else if (reset) c_enable = 1'b?;
        else c_enable = 0;
        
    end
    
    always @(*) begin
        if ((Q == 12 & enable) | reset) begin
            c_load = 1;
            c_d = 4'b0001;
        end
        else begin
            c_load = 0;
            c_d = 4'bzzzz;
        end
   end
       
 endmodule


```
